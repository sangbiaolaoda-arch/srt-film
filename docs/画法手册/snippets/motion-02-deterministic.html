这个片段演示什么：动画不自己走时间，而是 由外部给定一个 t。拖动滑块设定 t，画面就是 f(t) 的结果 —— 同一个 t 反复拖过去，渲染出的数值和画面完全一致。

为什么这里不用 GSAP？ 这种「逐帧可复现」骨架的重点不是库，而是一个模式：动画暂停，时间由外部给定（GSAP 里就是 timeline({paused:true}) + tl.time(t)）。下面用原生 JS 手写了一个约 20 行的等效时间轴来演示这个模式，所以本页不引任何 CDN。真实项目里两者可以互换。
核心代码 
// —— 约 20 行的等效时间轴：只做「给定 t 就渲染那一帧」，自己绝不走时间 ——
function createTimeline() {
  const items = [];
  let duration = 0;
  return {
    to(tween) {
      const start = tween.at ?? duration;          // 不写 at 就接在末尾
      items.push(Object.assign({ start }, tween));
      duration = Math.max(duration, start + tween.duration);
      return this;
    },
    get duration() { return duration; },
    time(t) {                                       // 关键点：手动 seek，纯函数式渲染
      for (const it of items) {
        const p = Math.min(1, Math.max(0, (t - it.start) / it.duration));
        const e = it.ease ? it.ease(p) : p;
        it.apply(it.from + (it.to - it.from) * e);
      }
      return this;
    }
  };
}

// 缓动只依赖 p，不依赖真实时间
const powerOut = (p) => 1 - Math.pow(1 - p, 3);
const backOut  = (p) => 1 + 2.70158 * Math.pow(p - 1, 3) + 1.70158 * Math.pow(p - 1, 2);

const tl = createTimeline();
tl.to({ at: 0,    duration: 1.0, ease: powerOut, from: 0,   to: 300,
        apply: (v) => { box1.style.transform = "translateX(" + v + "px)"; } })
  .to({ at: 0.55, duration: 1.5, ease: (p) => p,  from: 0,   to: 360,
        apply: (v) => { box2.style.transform = "rotate(" + v + "deg)"; } })
  .to({ at: 1.1,  duration: 1.0, ease: backOut,  from: 0.4, to: 1,
        apply: (v) => { box3.style.transform = "scale(" + v + ")"; } });

// 拖动 = 设定 t；同一 t 永远渲染出同一帧
slider.addEventListener("input", () => tl.time(parseFloat(slider.value)));
