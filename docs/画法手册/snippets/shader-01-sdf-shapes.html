WebGL 距离场画形状

这个片段演示什么：不去「画一个圆」，而是对每个像素回答「你到圆心的距离是多少」，再用 step / smoothstep 把距离切成形状。用 fwidth 做像素级自适应抗锯齿。移动鼠标，四个形状会跟着走。

① 四个形状，一个 fragment shader

屏幕分成 2×2 格，每格一个形状，全部用 SDF（有符号距离场）写：d < 0 在形状内部，d > 0 在外部。

移动鼠标 → 形状跟随
当前浏览器/环境不支持 WebGL2，无法运行此片段。
请用较新版本的 Chrome / Edge / Firefox / Safari 打开。
左上 · 十字 = 两个矩形 SDF 取 min（并集）
右上 · 星形 = 极坐标下 cos(a·5) 调制半径
左下 · 圆 = length(p) - r
右下 · 矩形 = abs(p) - b 的 sdBox
核心代码 
// ---- 顶点着色器：一个铺满屏幕的四边形 ----
#version 300 es
in vec2 aPos;
void main() { gl_Position = vec4(aPos, 0.0, 1.0); }

// ---- 片段着色器：把图形当距离场画 ----
#version 300 es
precision highp float;
uniform vec2 uResolution;
uniform vec2 uMouse;          // 0..1
out vec4 fragColor;

// 矩形 SDF：p 为相对中心的坐标，b 为半宽半高
float sdBox(vec2 p, vec2 b) {
  vec2 d = abs(p) - b;
  return length(max(d, 0.0)) + min(max(d.x, d.y), 0.0);
}

void main() {
  vec2 st = gl_FragCoord.xy / uResolution;   // 像素坐标 → 0..1（原点左下）

  // 关键点：把鼠标位移叠加到形状位置上
  vec2 off = (uMouse - 0.5) * 0.30;

  // 2×2 分格
  int col = st.x > 0.5 ? 1 : 0;
  int row = st.y > 0.5 ? 1 : 0;              // 0 = 下半屏
  int idx = row * 2 + col;

  // 本格局部坐标 -0.5..0.5，并修正宽高比（否则圆变椭圆）
  vec2 q = vec2(st.x - float(col) * 0.5, st.y - float(row) * 0.5) * 2.0 - 0.5;
  q.x *= uResolution.x / uResolution.y;
  vec2 p = q - off;

  float d;
  if (idx == 0)      d = length(p) - 0.26;                        // 圆
  else if (idx == 1) d = sdBox(p, vec2(0.30, 0.20));              // 矩形
  else if (idx == 2) d = min(sdBox(p, vec2(0.07, 0.32)),          // 十字 = 并集
                             sdBox(p, vec2(0.32, 0.07)));
  else {                                                          // 星
    float r = length(p);
    float a = atan(p.y, p.x);
    float k = 0.5 + 0.5 * cos(a * 5.0);
    d = r - mix(0.12, 0.32, k);
  }

  // 关键点：fwidth(d) ≈ 一个像素内 d 的变化量 → 过渡带正好一个像素宽
  float w = fwidth(d);
  float m = 1.0 - smoothstep(-w, w, d);

  vec3 bg = vec3(0.980, 0.976, 0.961);
  vec3 fg = idx == 0 ? vec3(0.784, 0.251, 0.122)
          : idx == 1 ? vec3(0.110, 0.106, 0.094)
          : idx == 2 ? vec3(0.247, 0.435, 0.420)
                     : vec3(0.725, 0.541, 0.149);
  fragColor = vec4(mix(bg, fg, m), 1.0);     // 关键点：mix 把「形状值」变「颜色」
}
