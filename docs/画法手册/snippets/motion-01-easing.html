缓动曲线可视化

这个片段演示什么：8 种缓动函数画成「时间 → 进度」曲线，再让 8 个小球用各自曲线跑同一条赛道 —— 曲线形状和图上的快慢一眼就能对上。

① 曲线：横轴是时间，纵轴是完成进度

纵轴超出 0~1 的部分就是「过冲」（back / elastic 靠它实现回弹）。注意 bounce / elastic 会反复穿越 0~1，所以没法用单条 cubic-bezier 表达。

② 赛跑：同一时长，不同缓动

所有小球时长相同，只是位移曲线不同。ease-out 起步最猛（立即响应），ease-in 起步最慢、末尾最快（适合退场）。

③ 什么时候用哪种

场景用哪种理由
元素进场（出现、弹出、淡入、滑入）ease-out起步快 = 立即响应，尾段慢 = 停在目标上显得「到位」
元素退场（关闭、消失、滑出）ease-in + 更短时长慢起快走，让元素迅速离开而不是赖着不走
纯位移 / A→B（滑轨、指示器切换）ease-in-out两头慢中间快，符合物体加速再减速的直觉
持续循环（spinner、进度、呼吸）linear首尾无缝衔接，循环最稳
物理回弹 / 果冻感back-out冲过终点再收回来，有质量感
机械 / 数字跳变 / 逐格steps()一格一格跳，不做平滑插值

两条硬规则：① 进场用 ease-out，退场用 ease-in；② 同一个动效里，进场的缓动 ≠ 退场的缓动。

核心代码 
// cubic-bezier(x1,y1,x2,y2) → 求值函数（牛顿迭代解 x(t)=x）
function cubicBezier(x1, y1, x2, y2) {
  const A = (a, b) => 1 - 3 * b + 3 * a;
  const B = (a, b) => 3 * b - 6 * a;
  const C = (a) => 3 * a;
  const calc = (t, a, b) => ((A(a, b) * t + B(a, b)) * t + C(a)) * t;
  return function (x) {
    if (x <= 0) return 0;
    if (x >= 1) return 1;
    let t = x;
    for (let i = 0; i < 8; i++) {
      const g = calc(t, x1, x2) - x;
      const d = 3 * A(x1, x2) * t * t + 2 * B(x1, x2) * t + C(x1);
      if (Math.abs(g) < 1e-6 || Math.abs(d) < 1e-6) break;
      t -= g / d;
    }
    return calc(t, y1, y2);
  };
}

const easings = [
  { name: "linear",      fn: (t) => t },
  { name: "ease",        fn: cubicBezier(0.25, 0.1, 0.25, 1) },
  { name: "ease-in",     fn: cubicBezier(0.42, 0, 1, 1) },
  { name: "ease-out",    fn: cubicBezier(0, 0, 0.58, 1) },
  { name: "ease-in-out", fn: cubicBezier(0.42, 0, 0.58, 1) },
  { name: "back-out",    fn: cubicBezier(0.34, 1.56, 0.64, 1) },
  // bounce / elastic 会多次穿越 0~1，只能用函数手写，无法用单条 bezier 表达
  { name: "elastic-out", fn: (t) => t <= 0 ? 0 : t >= 1 ? 1
      : Math.pow(2, -10 * t) * Math.sin((t * 10 - 0.75) * (2 * Math.PI / 3)) + 1 },
  { name: "bounce-out",  fn: (t) => {
      const n = 7.5625, d = 2.75;
      if (t < 1 / d) return n * t * t;
      if (t < 2 / d) { t -= 1.5 / d; return n * t * t + 0.75; }
      if (t < 2.5 / d) { t -= 2.25 / d; return n * t * t + 0.9375; }
      t -= 2.625 / d; return n * t * t + 0.984375;
  } }
];
